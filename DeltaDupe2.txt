--// ===== FUNCTIONS =====
local function applyTheme(color)
    -- Main
    Main.BackgroundColor3 = Color3.fromRGB(
        math.max(5, math.floor(color.R * 0.05)),
        math.max(5, math.floor(color.G * 0.05)),
        math.max(5, math.floor(color.B * 0.06))
    )
    MainCorner.Parent = Main
    
    -- Top
    Top.BackgroundColor3 = Color3.fromRGB(
        math.max(8, math.floor(color.R * 0.07)),
        math.max(8, math.floor(color.G * 0.07)),
        math.max(8, math.floor(color.B * 0.08))
    )
    
    -- Sidebar
    Sidebar.BackgroundColor3 = Color3.fromRGB(
        math.max(7, math.floor(color.R * 0.06)),
        math.max(7, math.floor(color.G * 0.06)),
        math.max(7, math.floor(color.B * 0.08))
    )
    
    -- Execute button
    Execute.BackgroundColor3 = color
    
    -- Size fill
    SizeFill.BackgroundColor3 = color
    
    -- Color preview
    ColorPreview.BackgroundColor3 = color
    
    -- Tab active color
    if TradeContent.Visible then
        TradeTab.BackgroundColor3 = color
    end
    if SettingsContent.Visible then
        SettingsTab.BackgroundColor3 = color
    end
    
    -- Stroke color
    Stroke.Color = Color3.fromRGB(
        math.min(255, color.R + 30),
        math.min(255, color.G + 30),
        math.min(255, color.B + 30)
    )
end

    SizeLabel.Text = "UI Size: " .. math.floor(scale * 100) .. "%"
    SizeFill.Size = UDim2.new(scale, 0, 1, 0)
end

--// ===== ITEMS =====
local Items = {
    { display = "Gold50g", search = {"Gold50g"} },
    { display = "Rubles", search = {"Rubles"} },
    { display = "FlareGun", search = {"FlareGun"} },
    { display = "7.62x39 Tracer", search = {"7.62x39 Tracer Bullet", "7.62x39 Tracer"} }
}
local CurrentItem = 1

local function findItem(itemName)
    local Inventory = RSP.Inventory.WastelandShirt.Inventory
    return Inventory:FindFirstChild(itemName)
end

Item.MouseButton1Click:Connect(function()
    CurrentItem = CurrentItem % #Items + 1
    if CurrentItem == 0 then CurrentItem = 1 end
    local data = Items[CurrentItem]
    Item.Text = data.display
    Status.Text = "●  Selected: " .. data.display
    Status.TextColor3 = Color3.fromRGB(100, 190, 125)
end)

Execute.MouseButton1Click:Connect(function()
    local amount = tonumber(Amount.Text)
    if not amount or amount < 1 then
        Status.Text = "●  Invalid amount"
        Status.TextColor3 = Color3.fromRGB(220, 100, 100)
        return
    end
    amount = math.floor(amount)
    
    local data = Items[CurrentItem]
    local foundItem = nil
    
    for _, name in ipairs(data.search) do
        foundItem = findItem(name)
        if foundItem then break end
    end
    
    if not foundItem then
        Status.Text = "●  " .. data.display .. " not found"
        Status.TextColor3 = Color3.fromRGB(220, 100, 100)
        return
    end
    
    local Trade = RS.Remotes:FindFirstChild("Trade")
    if not Trade then
        Status.Text = "●  Trade remote missing"
        Status.TextColor3 = Color3.fromRGB(220, 100, 100)
        return
    end
    
    local tradeList = {}
    for i = 1, amount do
        table.insert(tradeList, foundItem)
    end
    
    local success, result = pcall(function()
        return Trade:InvokeServer({Action = "Update", TradeList = tradeList})
    end)
    
    if not success then
        Status.Text = "●  Update failed"
        Status.TextColor3 = Color3.fromRGB(220, 100, 100)
        return
    end
    
    local confirmSuccess, confirmResult = pcall(function()
        return Trade:InvokeServer({Action = "Confirm"})
    end)
    
    if confirmSuccess then
        Status.Text = "●  Executed " .. amount .. "x " .. data.display
        Status.TextColor3 = Color3.fromRGB(100, 190, 125)
    else
        Status.Text = "●  Confirm failed"
        Status.TextColor3 = Color3.fromRGB(220, 100, 100)
    end
end)

--// ===== DRAGGING =====
local dragging = false
local dragStart
local startPosition

Top.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or 
       input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPosition = Main.Position
    end
end)

Top.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or 
       input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or 
                     input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        Main.Position = UDim2.new(
            startPosition.X.Scale,
            startPosition.X.Offset + delta.X,
            startPosition.Y.Scale,
            startPosition.Y.Offset + delta.Y
        )
    end
end)
