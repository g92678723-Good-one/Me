--// ITEMS DATABASE
local Items = {
    { display = "Gold50g", search = {"Gold50g"} },
    { display = "Rubles", search = {"Rubles"} },
    { display = "FlareGun", search = {"FlareGun"} },
    { display = "7.62x39 Tracer", search = {"7.62x39 Tracer", "7.62x39 Tracer Bullet"} }
}
local CurrentItem = 1

local function findItem(itemName)
    local Inventory = RSP:FindFirstChild("Inventory")
    if not Inventory then
        print("[HUB] Inventory not found in RSP")
        return nil
    end
    
    local WastelandShirt = Inventory:FindFirstChild("WastelandShirt")
    if not WastelandShirt then
        print("[HUB] WastelandShirt not found")
        return nil
    end
    
    local InventoryFolder = WastelandShirt:FindFirstChild("Inventory")
    if not InventoryFolder then
        print("[HUB] Inventory folder not found in WastelandShirt")
        return nil
    end
    
    local found = InventoryFolder:FindFirstChild(itemName)
    if found then return found end
    
    return nil
end

Item.MouseButton1Click:Connect(function()
    CurrentItem = CurrentItem % #Items + 1
    if CurrentItem == 0 then CurrentItem = 1 end
    local data = Items[CurrentItem]
    Item.Text = data.display
    Status.Text = "●  Selected: " .. data.display
    Status.TextColor3 = Color3.fromRGB(100, 190, 125)
end)

Execute.MouseButton1Click:Connect(function()
    local amount = tonumber(Amount.Text)
    if not amount or amount < 1 then
        Status.Text = "●  Invalid amount"
        Status.TextColor3 = Color3.fromRGB(220, 100, 100)
        return
    end
    amount = math.floor(amount)
    
    local data = Items[CurrentItem]
    local foundItem = nil
    
    for _, name in ipairs(data.search) do
        foundItem = findItem(name)
        if foundItem then
            print("[HUB] Found item: " .. name)
            break
        end
    end
    
    if not foundItem then
        Status.Text = "●  " .. data.display .. " not found"
        Status.TextColor3 = Color3.fromRGB(220, 100, 100)
        print("[HUB] Could not find any variant of: " .. data.display)
        return
    end
    
    local Trade = RS.Remotes:FindFirstChild("Trade")
    if not Trade then
        Status.Text = "●  Trade remote missing"
        Status.TextColor3 = Color3.fromRGB(220, 100, 100)
        print("[HUB] Trade remote not found in RS.Remotes")
        return
    end
    
    local tradeList = {}
    for i = 1, amount do
        table.insert(tradeList, foundItem)
    end
    
    local success, result = pcall(function()
        return Trade:InvokeServer({Action = "Update", TradeList = tradeList})
    end)
    
    if not success then
        Status.Text = "●  Update failed: " .. tostring(result)
        Status.TextColor3 = Color3.fromRGB(220, 100, 100)
        print("[HUB] Update error: " .. tostring(result))
        return
    end
    
    local confirmSuccess, confirmResult = pcall(function()
        return Trade:InvokeServer({Action = "Confirm"})
    end)
    
    if confirmSuccess then
        Status.Text = "●  Executed " .. amount .. "x " .. data.display
        Status.TextColor3 = Color3.fromRGB(100, 190, 125)
    else
        Status.Text = "●  Confirm failed"
        Status.TextColor3 = Color3.fromRGB(220, 100, 100)
        print("[HUB] Confirm error: " .. tostring(confirmResult))
    end
end)

--// SIZE CONTROLS
local function updateUISize(scale)
    scale = math.max(0.6, math.min(1.5, scale))
    uiScale = scale
    
    local baseW = 360 * scale
    local baseH = 230 * scale
    
    Main.Size = UDim2.new(0, baseW, 0, baseH)
    Main.Position = UDim2.new(0.5, -baseW/2, 0.5, -baseH/2)
    
    SizeLabel.Text = "UI Size: " .. math.floor(scale * 100) .. "%"
    SizeFill.Size = UDim2.new(scale / 1.5, 0, 1, 0)
end

SizeMinus.MouseButton1Click:Connect(function()
    updateUISize(uiScale - 0.1)
end)

SizePlus.MouseButton1Click:Connect(function()
    updateUISize(uiScale + 0.1)
end)

--// DRAGGING
local dragging = false
local dragStart
local startPosition

Top.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        dragStart = input.Position
        startPosition = Main.Position
    end
end)

Top.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = false
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
        local delta = input.Position - dragStart
        Main.Position = UDim2.new(
            startPosition.X.Scale,
            startPosition.X.Offset + delta.X,
            startPosition.Y.Scale,
            startPosition.Y.Offset + delta.Y
        )
    end
end)
