--// ===== ITEMS FOR TRADE =====
local Items = {
    { display = "Gold50g", search = {"Gold50g"} },
    { display = "Rubles", search = {"Rubles"} },
    { display = "FlareGun", search = {"FlareGun"} },
    { display = "7.62x39", search = {"7.62x39T", "7.62.39T", "7.62x39", "762x39T", "762.39T"} }
}
local CurrentItem = 1

local function findItemInInventory(itemName)
    local Inventory = RSP.Inventory.WastelandShirt.Inventory
    local found = Inventory:FindFirstChild(itemName)
    if found then return found end
    
    for _, child in ipairs(Inventory:GetChildren()) do
        if string.lower(child.Name) == string.lower(itemName) then
            return child
        end
    end
    
    for _, child in ipairs(Inventory:GetChildren()) do
        local cleanChild = string.gsub(child.Name, "%.", "")
        local cleanItem = string.gsub(itemName, "%.", "")
        if string.lower(cleanChild) == string.lower(cleanItem) then
            return child
        end
    end
    
    return nil
end

Item.MouseButton1Click:Connect(function()
    CurrentItem = CurrentItem % #Items + 1
    local currentData = Items[CurrentItem]
    Item.Text = currentData.display
    Status.Text = "●  Selected: " .. currentData.display
    Status.TextColor3 = Color3.fromRGB(100, 190, 125)
end)

Execute.MouseButton1Click:Connect(function()
    local amount = tonumber(Amount.Text)
    if not amount or amount < 1 then
        Status.Text = "●  Invalid amount"
        Status.TextColor3 = Color3.fromRGB(220, 100, 100)
        return
    end
    amount = math.floor(amount)
    
    local currentData = Items[CurrentItem]
    local foundItem = nil
    for _, searchName in ipairs(currentData.search) do
        foundItem = findItemInInventory(searchName)
        if foundItem then break end
    end
    
    if not foundItem then
        Status.Text = "●  " .. currentData.display .. " not found"
        Status.TextColor3 = Color3.fromRGB(220, 100, 100)
        return
    end
    
    local Trade = RS.Remotes:FindFirstChild("Trade")
    if not Trade then
        Status.Text = "●  Trade remote missing"
        Status.TextColor3 = Color3.fromRGB(220, 100, 100)
        return
    end
    
    local tradeList = {}
    for i = 1, amount do
        table.insert(tradeList, foundItem)
    end
    
    local success, result = pcall(function()
        return Trade:InvokeServer({Action = "Update", TradeList = tradeList})
    end)
    
    if not success then
        Status.Text = "●  Update failed"
        Status.TextColor3 = Color3.fromRGB(220, 100, 100)
        return
    end
    
    local confirmSuccess, confirmResult = pcall(function()
        return Trade:InvokeServer({Action = "Confirm"})
    end)
    
    if confirmSuccess then
        Status.Text = "●  Executed " .. amount .. "x " .. currentData.display
        Status.TextColor3 = Color3.fromRGB(100, 190, 125)
    else
        Status.Text = "●  Confirm failed"
        Status.TextColor3 = Color3.fromRGB(220, 100, 100)
    end
end)

--// ===== DRAGGING =====
local dragging = false
local dragStart
local startPosition

Top.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPosition = Main.Position
    end
end)

Top.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        Main.Position = UDim2.new(
            startPosition.X.Scale,
            startPosition.X.Offset + delta.X,
            startPosition.Y.Scale,
            startPosition.Y.Offset + delta.Y
        )
    end
end)
