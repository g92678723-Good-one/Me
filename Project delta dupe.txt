--// Items
local Items = {
    "Gold50g",
    "Rubles",
    "FlareGun"
}

local CurrentItem = 1

--// Cycle item
Item.MouseButton1Click:Connect(function()
    CurrentItem += 1

    if CurrentItem > #Items then
        CurrentItem = 1
    end

    Item.Text = Items[CurrentItem]
    Status.Text = "●  Selected: " .. Items[CurrentItem]
end)

--// Execute trade - FIXED DUPE LOGIC
Execute.MouseButton1Click:Connect(function()
    local amount = tonumber(Amount.Text)

    if not amount or amount < 1 then
        Status.Text = "●  Invalid amount"
        Status.TextColor3 = Color3.fromRGB(220, 100, 100)
        return
    end

    amount = math.floor(amount)
    local itemName = Items[CurrentItem]

    --// Find item in inventory - check multiple paths
    local Inventory = RSP:FindFirstChild("Inventory") or RSP.Inventory.WastelandShirt.Inventory
    local ItemX = Inventory:FindFirstChild(itemName)

    if not ItemX then
        Status.Text = "●  " .. itemName .. " not found"
        Status.TextColor3 = Color3.fromRGB(220, 100, 100)
        return
    end

    local Trade = RS.Remotes:FindFirstChild("Trade")

    if not Trade then
        Status.Text = "●  Trade remote missing"
        Status.TextColor3 = Color3.fromRGB(220, 100, 100)
        return
    end

    --// Build trade list - reference same item N times
    local tradeList = {}
    for i = 1, amount do
        table.insert(tradeList, ItemX)
    end

    --// CRITICAL FIX: Wait between update and confirm
    local success, result = pcall(function()
        return Trade:InvokeServer({
            Action = "Update",
            TradeList = tradeList,
            Item = itemName
        })
    end)

    if not success then
        Status.Text = "●  Update failed"
        Status.TextColor3 = Color3.fromRGB(220, 100, 100)
        warn("Update error: " .. tostring(result))
        return
    end

    --// TIMING FIX: Small delay ensures server processes update
    wait(0.15)

    local confirmSuccess, confirmResult = pcall(function()
        return Trade:InvokeServer({
            Action = "Confirm",
            Item = itemName,
            Amount = amount
        })
    end)

    if confirmSuccess then
        Status.Text = "●  Duped " .. amount .. "x " .. itemName
        Status.TextColor3 = Color3.fromRGB(100, 190, 125)
    else
        Status.Text = "●  Confirm failed"
        Status.TextColor3 = Color3.fromRGB(220, 100, 100)
        warn("Confirm error: " .. tostring(confirmResult))
    end
end)

--// Dragging
local dragging = false
local dragStart
local startPosition

Top.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        dragging = true
        dragStart = input.Position
        startPosition = Main.Position
    end
end)

Top.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        dragging = false
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (
        input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch
    ) then

        local delta = input.Position - dragStart

        Main.Position = UDim2.new(
            startPosition.X.Scale,
            startPosition.X.Offset + delta.X,
            startPosition.Y.Scale,
            startPosition.Y.Offset + delta.Y
        )
    end
end)
